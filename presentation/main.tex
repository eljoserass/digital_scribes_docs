\begin{frame}[t,noframenumbering]{Opciones para definir una evaluación}
\topic{Apoyo A1 · protocolo pendiente}
\begin{itemize}
\item \textbf{Casos comunes:} audio o simulaciones, contexto equivalente y revisión clínica independiente.
\item \textbf{Efecto asistencial:} comparar el flujo habitual y el uso del escriba; antes/después o grupo contemporáneo según viabilidad.

\item \textbf{Decisiones previas:} muestra, formación, versiones y criterios de continuidad.
\end{itemize}
\takeaway{Un banco de consultas compara notas; el ahorro asistencial requiere observar el trabajo real.}
